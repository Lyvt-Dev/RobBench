cmake_minimum_required(VERSION 3.25)
project(RobBench VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ROBBENCH_BUILD_TESTS "Build unit and integration tests" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
find_package(Vulkan 1.3 REQUIRED)
find_program(ROBBENCH_GLSLANG NAMES glslangValidator
    HINTS "${Vulkan_GLSLANG_VALIDATOR_EXECUTABLE}" "$ENV{VULKAN_SDK}/Bin"
          "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/tools/glslang"
    REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_path(VMA_INCLUDE_DIR vk_mem_alloc.h PATH_SUFFIXES vma REQUIRED)
add_library(robbench_core src/core/Project.cpp src/modeling/TransformGizmo.cpp src/rigging/Rigs.cpp src/animation/Animation.cpp src/io/Serialization.cpp)
target_include_directories(robbench_core PUBLIC src)
target_compile_definitions(robbench_core PUBLIC GLM_FORCE_RADIANS GLM_FORCE_DEPTH_ZERO_TO_ONE)
target_link_libraries(robbench_core PUBLIC glm::glm nlohmann_json::nlohmann_json Qt6::Core)
add_executable(RobBench src/main.cpp src/ui/MainWindow.cpp src/ui/Timeline.cpp src/ui/TransformToolbar.cpp src/ui/EditorSmokeTest.cpp src/renderer/Viewport.cpp src/renderer/VulkanRenderer.cpp)
target_include_directories(RobBench SYSTEM PRIVATE ${VMA_INCLUDE_DIR})
target_link_libraries(RobBench PRIVATE robbench_core Qt6::Widgets Vulkan::Vulkan spdlog::spdlog)
if(MSVC)
  target_compile_options(robbench_core PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(RobBench PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(robbench_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(RobBench PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
endif()
foreach(stage vert frag)
  set(output "${CMAKE_CURRENT_BINARY_DIR}/scene.${stage}.spv")
  add_custom_command(OUTPUT "${output}" COMMAND "${ROBBENCH_GLSLANG}" --target-env vulkan1.3 -V "${CMAKE_CURRENT_SOURCE_DIR}/resources/shaders/scene.${stage}" -o "${output}" DEPENDS "resources/shaders/scene.${stage}" VERBATIM)
  list(APPEND SHADER_BINARIES "${output}")
endforeach()
qt_add_resources(RobBench shaders PREFIX /shaders BASE "${CMAKE_CURRENT_BINARY_DIR}" FILES ${SHADER_BINARIES})
if(ROBBENCH_BUILD_TESTS)
  enable_testing()
  find_package(GTest CONFIG REQUIRED)
  add_executable(robbench_tests tests/CoreTests.cpp tests/GizmoTests.cpp)
  target_link_libraries(robbench_tests PRIVATE robbench_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(robbench_tests)
  option(ROBBENCH_VULKAN_TESTS "Run native editor smoke tests (requires display and Vulkan 1.3)" OFF)
  if(ROBBENCH_VULKAN_TESTS)
    foreach(rig R6 R15)
      add_test(NAME Editor.${rig} COMMAND RobBench --smoke-test "${CMAKE_CURRENT_BINARY_DIR}/smoke-${rig}" --smoke-rig ${rig})
      set_tests_properties(Editor.${rig} PROPERTIES TIMEOUT 30 RESOURCE_LOCK vulkan_display FAIL_REGULAR_EXPRESSION "VUID-|Validation Error")
    endforeach()
  endif()
endif()
include(GNUInstallDirs)
install(TARGETS RobBench RUNTIME_DEPENDENCY_SET robbench_runtime
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} BUNDLE DESTINATION .)
if(WIN32)
  install(RUNTIME_DEPENDENCY_SET robbench_runtime
    DIRECTORIES "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/bin"
    PRE_EXCLUDE_REGEXES "api-ms-.*" "ext-ms-.*"
    POST_EXCLUDE_REGEXES [=[.*[Ww][Ii][Nn][Dd][Oo][Ww][Ss][/\\].*]=]
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ${CMAKE_INSTALL_BINDIR})
  include(InstallRequiredSystemLibraries)
endif()
if(DEFINED VCPKG_INSTALLED_DIR)
  foreach(dep qtbase vulkan vulkan-loader vulkan-headers vulkan-hpp vulkan-memory-allocator glm nlohmann-json spdlog fmt)
    set(notice "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${dep}/copyright")
    if(EXISTS "${notice}")
      install(FILES "${notice}" DESTINATION "${CMAKE_INSTALL_DATADIR}/RobBench/licenses/${dep}")
    endif()
  endforeach()
endif()
# Qt 6.5 renamed FILENAME_VARIABLE to OUTPUT_SCRIPT.
if(Qt6_VERSION VERSION_LESS 6.5)
  qt_generate_deploy_app_script(TARGET RobBench FILENAME_VARIABLE deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
else()
  qt_generate_deploy_app_script(TARGET RobBench OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
endif()
install(SCRIPT ${deploy_script})
install(DIRECTORY docs plugins DESTINATION ${CMAKE_INSTALL_DATADIR}/RobBench)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/RobBench)
set(CPACK_PACKAGE_NAME RobBench)
set(CPACK_PACKAGE_VENDOR "RobBench contributors")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
if(WIN32)
  set(CPACK_GENERATOR "ZIP;NSIS")
  set(CPACK_NSIS_DISPLAY_NAME "RobBench")
  set(CPACK_NSIS_EXECUTABLES_DIRECTORY "bin")
  set(CPACK_NSIS_MENU_LINKS "bin/RobBench.exe" "RobBench")
else()
  set(CPACK_GENERATOR TGZ)
endif()
include(CPack)
